\esercizio{Esercizio 1 -- Problema, parte sperimentale}{}{}{}{}{}
\begin{enun}
Una sfera di massa $\SI{0.50}{kg}$ cade in un fluido con resistenza $-bv$, $b=\SI{2.0}{kg/s}$. Sono misurate le velocità $v=0{,}81;\,1{,}35;\,1{,}95;\,2{,}23;\,2{,}35\ \si{m/s}$ agli istanti $t=0{,}1;\,0{,}2;\,0{,}4;\,0{,}6;\,0{,}8\ \si{s}$. (a) Determina la velocità limite. (b) Rappresenta i dati $v$--$t$ e individua l'andamento. (c) Confronta i dati con $v(t)=\frac{mg}{b}\left(1-e^{-bt/m}\right)$ e stima la costante di tempo. (d) Discuti la compatibilità fra dati e modello.
\end{enun}
\tabpuntiesame{ANA}{SVI}{DAT}{ARG}{60}{60}{180}{100}{400}
\evidenza{E01}{1}{ANA}{VEL}{60}
\begin{boxana}[Riconosce la condizione di velocità limite]
\begin{enumerate}
\item A velocità limite l'accelerazione è nulla: la forza peso uguaglia la resistenza, $mg=bv_\ell$.
\end{enumerate}
\end{boxana}
\evidenza{E02}{1}{SVI}{VEL}{60}
\begin{boxsvi}[Calcola la velocità limite]
\begin{enumerate}
\item $v_\ell=\dfrac{mg}{b}=\dfrac{\SI{0.50}{kg}\cdot\SI{9.81}{m/s^2}}{\SI{2.0}{kg/s}}=\SI{2.45}{m/s}$.
\end{enumerate}
\end{boxsvi}
\evidenza{E03}{1}{DAT}{VEL}{80}
\begin{boxdat}[Rappresenta i dati e individua la saturazione]
\begin{enumerate}
\item Grafico $v$--$t$ con assi, unità e scala.
\item I dati crescono sempre più lentamente e tendono a $\SI{2.45}{m/s}$.
\end{enumerate}
\end{boxdat}
\evidenza{E04}{1}{DAT}{VEL}{100}
\begin{boxdat}[Confronta i dati con il modello e stima $\tau$]
\begin{enumerate}
\item $\tau=m/b=\SI{0.25}{s}$.
\item Per ogni $t$ misurato il modello dà $0{,}81;\,1{,}35;\,1{,}95;\,2{,}23;\,2{,}35\ \si{m/s}$, uguali ai dati entro $\SI{0.01}{m/s}$.
\end{enumerate}
\end{boxdat}
\evidenza{E05}{1}{ARG}{VEL}{100}
\begin{boxarg}[Valuta la compatibilità fra dati e modello]
\begin{enumerate}
\item Le differenze sono entro l'incertezza di lettura: dati e modello sono compatibili.
\item Cause di discrepanza: tempo di reazione, resistenza non esattamente lineare.
\end{enumerate}
\end{boxarg}
\esercizio{Esercizio 2 -- Problema, parte teorica}{}{}{}{}{}
\begin{enun}
Con $v(t)=\frac{mg}{b}\left(1-e^{-bt/m}\right)$: (a) ricava l'accelerazione $a(t)$ e interpreta $a(0)$; (b) imposta e calcola lo spazio percorso nel primo secondo; (c) rappresenta $s(t)$ e confrontalo con la caduta libera; (d) giustifica il comportamento per $t\to\infty$ e la coerenza fisica.
\end{enun}
\tabpuntiesame{ANA}{SVI}{DAT}{ARG}{190}{240}{70}{100}{600}
\evidenza{E06}{2}{ANA}{SPA}{90}
\begin{boxana}[Collega accelerazione e derivata di $v$]
\begin{enumerate}
\item $a(t)=\dfrac{dv}{dt}$; $a(0)$ è l'accelerazione iniziale quando la resistenza è nulla.
\end{enumerate}
\end{boxana}
\evidenza{E07}{2}{SVI}{SPA}{120}
\begin{boxsvi}[Calcola $a(t)$ e $a(0)$]
\begin{enumerate}
\item $a(t)=g\,e^{-bt/m}$.
\item $a(0)=g=\SI{9.81}{m/s^2}$.
\end{enumerate}
\end{boxsvi}
\evidenza{E08}{2}{SVI}{SPA}{120}
\begin{boxsvi}[Integra $v(t)$ per lo spazio]
\begin{enumerate}
\item $s(t)=\dfrac{mg}{b}\left[t-\tau\left(1-e^{-t/\tau}\right)\right]$.
\item $s(\SI{1}{s})=\SI{2.45}{m/s}\cdot\left[\SI{1}{s}-\SI{0.25}{s}\cdot(1-e^{-4})\right]=\SI{1.85}{m}$.
\end{enumerate}
\end{boxsvi}
\evidenza{E09}{2}{DAT}{SPA}{70}
\begin{boxdat}[Rappresenta $s(t)$ e confronta con la caduta libera]
\begin{enumerate}
\item Grafico di $s(t)$ crescente e quasi lineare per $t\gg\tau$.
\item In caduta libera si avrebbe $\tfrac12gt^2=\SI{4.9}{m}$: la resistenza riduce molto lo spazio.
\end{enumerate}
\end{boxdat}
\evidenza{E10}{2}{ARG}{SPA}{100}
\begin{boxarg}[Giustifica il limite e la coerenza fisica]
\begin{enumerate}
\item Per $t\to\infty$ è $a\to0$ e $v\to v_\ell$: la velocità non supera $\frac{mg}{b}$.
\item Lo spazio nel primo secondo è minore di quello in caduta libera, come atteso.
\end{enumerate}
\end{boxarg}
\evidenza{E11}{2}{ANA}{SPA}{100}
\begin{boxana}[Imposta lo spazio come integrale]
\begin{enumerate}
\item $s(t)=\int_0^tv(t')\,dt'$ con $s(0)=0$.
\end{enumerate}
\end{boxana}
